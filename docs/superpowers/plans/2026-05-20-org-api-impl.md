# Org API — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rust/Axum REST API exposing the WF Engine Organization hierarchy and all 9 ORGTRVLANG traversal tokens backed by PostgreSQL.

**Architecture:** Layered modules — `handlers/` parses HTTP, `traversal/` owns DSL parsing and DB dispatch, `models.rs` holds domain types. Traversal handler derives `orgt_id` from the anchor ORGU so callers only supply the DSL expression. The existing `init.sql` schema and SQL functions are used as-is.

**Tech Stack:** Rust (stable), Axum 0.7, SQLx 0.7 (postgres + runtime-tokio-rustls), Tokio, Serde, thiserror, dotenvy, tracing.

```rust
use axum::{
    http::StatusCode,
    response::{IntoResponse, Response},
    Json,

use serde_json::json;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum AppError {
    #[error("{0}")]
    NotFound(String),
    #[error("{0}")]
    BadRequest(String),
    #[error(transparent)]
    Database(#[from] sqlx::Error),
}

impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let (status, message) = match &self {
            AppError::NotFound(msg) => (StatusCode::NOT_FOUND, msg.clone()),
            AppError::BadRequest(msg) => (StatusCode::BAD_REQUEST, msg.clone()),
            AppError::Database(e) => {
                tracing::error!("database error: {e}");
                (
                    StatusCode::INTERNAL_SERVER_ERROR,
                    "internal database error".to_string(),
                )
            }
        };
        (status, Json(json!({ "error": message }))).into_response()
    }
}
```

```rust
pub fn parse(expr: &str) -> Result<TraversalExpr, ParseError> {
    let expr = expr.trim();

    // children[T].children[T2]
    if let Some(rest) = expr.strip_prefix("children[") {
        if let Some(dot_pos) = rest.find("].children[") {
            let parent_type = &rest[..dot_pos];
            let after = &rest[dot_pos + "].children[".len()..];
            let child_type = after
                .strip_suffix(']')
                .ok_or_else(|| ParseError::Unknown(expr.to_string()))?;
            return Ok(TraversalExpr::ChildrenTChildrenT {
                parent_type: parent_type.to_string(),
                child_type:  child_type.to_string(),
            });
        }
        // children[T]
        let t = rest
            .strip_suffix(']')
            .ok_or_else(|| ParseError::Unknown(expr.to_string()))?;
        return Ok(TraversalExpr::ChildrenT(t.to_string()));
    }

    // up[T].children[T2]  |  up[T].children  |  up[T]
    if let Some(rest) = expr.strip_prefix("up[") {
        if let Some(dot_pos) = rest.find("].children") {
            let ancestor_type = &rest[..dot_pos];
            let after_children = &rest[dot_pos + "].children".len()..];
            if after_children.is_empty() {
                return Ok(TraversalExpr::UpTChildren(ancestor_type.to_string()));
            }
            if let Some(inner) = after_children
                .strip_prefix('[')
                .and_then(|s| s.strip_suffix(']'))
            {
                return Ok(TraversalExpr::UpTChildrenT {
                    ancestor_type: ancestor_type.to_string(),
                    child_type:    inner.to_string(),
                });
            }
            return Err(ParseError::Unknown(expr.to_string()));
        }
        // up[T]
        let t = rest
            .strip_suffix(']')
            .ok_or_else(|| ParseError::Unknown(expr.to_string()))?;
        return Ok(TraversalExpr::UpT(t.to_string()));
    }

    // siblings[T]
    if let Some(rest) = expr.strip_prefix("siblings[") {
        let t = rest
            .strip_suffix(']')
            .ok_or_else(|| ParseError::Unknown(expr.to_string()))?;
        return Ok(TraversalExpr::SiblingsT(t.to_string()));
    }

    match expr {
        "self"     => Ok(TraversalExpr::Self_),
        "parent"   => Ok(TraversalExpr::Parent),
        "siblings" => Ok(TraversalExpr::Siblings),
        "children" => Ok(TraversalExpr::Children),
        _          => Err(ParseError::Unknown(expr.to_string())),
    }
}
```

```rust
use sqlx::PgPool;
use uuid::Uuid;

use crate::{error::AppError, models::Orgu};
use super::parser::TraversalExpr;

const COLS: &str = "orgu_id, orgtnt_id, orgt_id, parent_orgu_id, \
                    path::text AS path, orgu_t, name, metadata, \
                    is_active, created_at, updated_at";

pub async fn execute(
    pool:    &PgPool,
    orgu_id: Uuid,
    orgt_id: Uuid,
    expr:    &TraversalExpr,
) -> Result<Vec<Orgu>, AppError> {
    macro_rules! q2 {
        ($fn:literal) => {
            sqlx::query_as::<_, Orgu>(
                &format!("SELECT {COLS} FROM {}($1, $2)", $fn)
            )
            .bind(orgu_id)
            .bind(orgt_id)
            .fetch_all(pool)
            .await
            .map_err(AppError::Database)
        };
    }
    macro_rules! q3 {
        ($fn:literal, $t:expr) => {
            sqlx::query_as::<_, Orgu>(
                &format!("SELECT {COLS} FROM {}($1, $2, $3)", $fn)
            )
            .bind(orgu_id)
            .bind(orgt_id)
            .bind($t)
            .fetch_all(pool)
            .await
            .map_err(AppError::Database)
        };
    }
    macro_rules! q4 {
        ($fn:literal, $t1:expr, $t2:expr) => {
            sqlx::query_as::<_, Orgu>(
                &format!("SELECT {COLS} FROM {}($1, $2, $3, $4)", $fn)
            )
            .bind(orgu_id)
            .bind(orgt_id)
            .bind($t1)
            .bind($t2)
            .fetch_all(pool)
            .await
            .map_err(AppError::Database)
        };
    }

    match expr {
        TraversalExpr::Self_    => q2!("orgtrvlang_self"),
        TraversalExpr::Parent   => q2!("orgtrvlang_parent"),
        TraversalExpr::Siblings => q2!("orgtrvlang_siblings"),
        TraversalExpr::SiblingsT(t)  => q3!("orgtrvlang_siblings_t", t),
        TraversalExpr::Children      => q2!("orgtrvlang_children"),
        TraversalExpr::ChildrenT(t)  => q3!("orgtrvlang_children_t", t),
        TraversalExpr::UpT(t)        => q3!("orgtrvlang_up_t", t),
        TraversalExpr::UpTChildren(t) => q3!("orgtrvlang_up_t_children", t),
        TraversalExpr::UpTChildrenT { ancestor_type, child_type } =>
            q4!("orgtrvlang_up_t_children_t", ancestor_type, child_type),
        TraversalExpr::ChildrenTChildrenT { parent_type, child_type } =>
            q4!("orgtrvlang_children_t_children_t", parent_type, child_type),
    }
}
```

```rust
use crate::{error::AppError, models::Orgtnt};

pub async fn list_orgtnt(
    State(pool): State<PgPool>,
) -> Result<Json<Vec<Orgtnt>>, AppError> {
    let rows = sqlx::query_as::<_, Orgtnt>(
        "SELECT orgtnt_id, name, code, is_active, created_at, updated_at
         FROM orgtnt
         ORDER BY name",
    )
    .fetch_all(&pool)
    .await?;
    Ok(Json(rows))
}

pub async fn get_orgtnt(
    State(pool): State<PgPool>,
    Path(id): Path<Uuid>,
) -> Result<Json<Orgtnt>, AppError> {
    sqlx::query_as::<_, Orgtnt>(
        "SELECT orgtnt_id, name, code, is_active, created_at, updated_at
         FROM orgtnt
         WHERE orgtnt_id = $1",
    )
    .bind(id)
    .fetch_optional(&pool)
    .await?
    .map(Json)
    .ok_or_else(|| AppError::NotFound(format!("orgtnt not found: {id}")))
}
```

```rust
use axum::{
    extract::{Path, Query, State},
    Json,
};
use serde::Deserialize;
use sqlx::PgPool;
use uuid::Uuid;

use crate::{
    error::AppError,
    models::Orgu,
    traversal::{executor, parser},
};

#[derive(Deserialize)]
pub struct TraverseQuery {
    pub expr: String,
}

pub async fn traverse_orgu(
    State(pool): State<PgPool>,
    Path(orgu_id): Path<Uuid>,
    Query(params): Query<TraverseQuery>,
) -> Result<Json<Vec<Orgu>>, AppError> {
    let expr = parser::parse(&params.expr)
        .map_err(|e| AppError::BadRequest(e.to_string()))?;

    let orgt_id: Uuid = sqlx::query_scalar::<_, Uuid>("SELECT orgt_id FROM orgu WHERE orgu_id = $1")
        .bind(orgu_id)
        .fetch_optional(&pool)
        .await?
        .ok_or_else(|| AppError::NotFound(format!("orgu not found: {orgu_id}")))?;

    let result = executor::execute(&pool, orgu_id, orgt_id, &expr).await?;
    Ok(Json(result))
}
```

```rust
mod config;
mod error;
mod handlers;
mod models;

use axum::{routing::get, Router};
use sqlx::postgres::PgPoolOptions;
use tracing_subscriber::{layer::SubscriberExt, util::SubscriberInitExt};

#[tokio::main]
async fn main() {
    tracing_subscriber::registry()
        .with(tracing_subscriber::EnvFilter::new(
            std::env::var("RUST_LOG").unwrap_or_else(|_| "info".into()),
        ))
        .with(tracing_subscriber::fmt::layer())
        .init();

    let config = config::Config::from_env().expect("config error");

    let pool = PgPoolOptions::new()
        .max_connections(10)
        .connect(&config.database_url)
        .await
        .expect("failed to connect to database");

    let app = Router::new()
        .route("/orgtnt",            get(handlers::orgtnt::list_orgtnt))
        .route("/orgtnt/:id",        get(handlers::orgtnt::get_orgtnt))
        .route("/orgtnt/:id/orgt",   get(handlers::orgt::list_orgt_by_tenant))
        .route("/orgt/:id/orgu",     get(handlers::orgu::list_orgu_by_tree))
        .route("/orgu/:id",          get(handlers::orgu::get_orgu))
        .route("/orgu/:id/traverse", get(handlers::traverse::traverse_orgu))
        .with_state(pool);

    let addr = format!("0.0.0.0:{}", config.port);
    tracing::info!("listening on {addr}");
    let listener = tokio::net::TcpListener::bind(&addr).await.unwrap();
    axum::serve(listener, app).await.unwrap();
```
