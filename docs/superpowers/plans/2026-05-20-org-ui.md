# Org UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-page Vite + React developer tool that renders the ORGT hierarchy with ReactFlow and lets the user run ORGTRVLANG traversal queries against the live org-api.

**Architecture:** 50/50 split layout — left panel shows tenant/tree dropdowns + ReactFlow canvas; right panel has a textarea query editor and results list. Shared state (query string, highlight IDs) lives in App. Each panel manages its own data-fetching state internally. Query format: `self = <uuid>` on line 1, traversal expression on line 2.

**Tech Stack:** Vite 5, React 18, TypeScript 5, @xyflow/react 12, @dagrejs/dagre, Vitest

```typescript
const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export type ParseResult =
  | { selfId: string; expr: string }
  | { error: string };

export function parseQuery(query: string): ParseResult {
  const lines = query.split('\n').map(l => l.trim()).filter(Boolean);

  const selfLine = lines[0] ?? '';
  const match = selfLine.match(/^self\s*=\s*(.+)$/i);
  if (!match) {
    return { error: 'İlk satır "self = <uuid>" formatında olmalı' };
  }

  const uuid = match[1].trim();
  if (!UUID_RE.test(uuid)) {
    return { error: `Geçersiz UUID: ${uuid}` };
  }

  const expr = lines[1] ?? '';
  if (!expr) {
    return { error: 'İkinci satırda traversal expression gerekli (örn: parent, children[branch])' };
  }

  return { selfId: uuid, expr };
}
```

```typescript
import type { Orgtnt, Orgt, Orgu } from './types';

const BASE = 'http://localhost:3000';

async function get<T>(url: string): Promise<T> {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json() as Promise<T>;
}

export function fetchTenants(): Promise<Orgtnt[]> {
  return get(`${BASE}/orgtnt`);
}

export function fetchTrees(orgtntId: string): Promise<Orgt[]> {
  return get(`${BASE}/orgtnt/${orgtntId}/orgt`);
}

export function fetchOrgus(orgtId: string): Promise<Orgu[]> {
  return get(`${BASE}/orgt/${orgtId}/orgu`);
}

export function traverse(orguId: string, expr: string): Promise<Orgu[]> {
  return get(`${BASE}/orgu/${orguId}/traverse?expr=${encodeURIComponent(expr)}`);
}
```

```typescript
import Dagre from '@dagrejs/dagre';
import type { Node, Edge } from '@xyflow/react';
import type { Orgu } from './types';

const NODE_W = 180;
const NODE_H = 48;

export function buildLayout(
  orgus: Orgu[],
  highlightIds: Set<string>,
): { nodes: Node[]; edges: Edge[] } {
  const g = new Dagre.graphlib.Graph();
  g.setDefaultEdgeLabel(() => ({}));
  g.setGraph({ rankdir: 'TB', ranksep: 50, nodesep: 25 });

  for (const o of orgus) {
    g.setNode(o.orgu_id, { width: NODE_W, height: NODE_H });
  }
  for (const o of orgus) {
    if (o.parent_orgu_id) g.setEdge(o.parent_orgu_id, o.orgu_id);
  }
  Dagre.layout(g);

  const nodes: Node[] = orgus.map(o => {
    const { x, y } = g.node(o.orgu_id);
    const highlighted = highlightIds.has(o.orgu_id);
    return {
      id: o.orgu_id,
      position: { x: x - NODE_W / 2, y: y - NODE_H / 2 },
      data: { label: `${o.name} [${o.orgu_t}]` },
      style: highlighted
        ? { background: '#fbbf24', color: '#1e293b', border: '2px solid #d97706', borderRadius: 6, fontSize: 11, padding: '4px 8px' }
        : { background: '#1e293b', color: '#e2e8f0', border: '1px solid #475569', borderRadius: 6, fontSize: 11, padding: '4px 8px' },
    };
  });

  const edges: Edge[] = orgus
    .filter(o => o.parent_orgu_id)
    .map(o => ({
      id: `e-${o.parent_orgu_id}-${o.orgu_id}`,
      source: o.parent_orgu_id!,
      target: o.orgu_id,
      style: { stroke: '#475569' },
    }));

  return { nodes, edges };
}
```

```tsx
  const [orgus, setOrgus] = useState<Orgu[]>([]);
  const [selectedTenantId, setSelectedTenantId] = useState('');
  const [selectedTreeId, setSelectedTreeId] = useState('');
  const [nodes, setNodes, onNodesChange] = useNodesState([]);
  const [edges, setEdges, onEdgesChange] = useEdgesState([]);
  const [loadError, setLoadError] = useState<string | null>(null);

  useEffect(() => {
    fetchTenants().then(setTenants).catch(e => setLoadError(String(e.message)));
  }, []);

  useEffect(() => {
    if (!selectedTenantId) return;
    setTrees([]);
    setSelectedTreeId('');
    setOrgus([]);
    fetchTrees(selectedTenantId).then(setTrees).catch(e => setLoadError(String(e.message)));
  }, [selectedTenantId]);

  useEffect(() => {
    if (!selectedTreeId) return;
    setOrgus([]);
    fetchOrgus(selectedTreeId).then(setOrgus).catch(e => setLoadError(String(e.message)));
  }, [selectedTreeId]);

  useEffect(() => {
    if (!orgus.length) { setNodes([]); setEdges([]); return; }
    const { nodes: n, edges: e } = buildLayout(orgus, highlightIds);
    setNodes(n);
    setEdges(e);
  }, [orgus, highlightIds]);

  const handleNodeClick = useCallback((_: React.MouseEvent, node: Node) => {
    onNodeClick(node.id);
  }, [onNodeClick]);

  return (
    <div style={{ height: '100%', display: 'flex', flexDirection: 'column' }}>
      <div style={{
        padding: '8px 12px',
        borderBottom: '1px solid #334155',
        display: 'flex',
        gap: 8,
        alignItems: 'center',
        flexShrink: 0,
      }}>
        <select value={selectedTenantId} onChange={e => setSelectedTenantId(e.target.value)} style={selectStyle}>
          <option value="">Tenant seç...</option>
          {tenants.map(t => <option key={t.orgtnt_id} value={t.orgtnt_id}>{t.name}</option>)}
        </select>
        <select
          value={selectedTreeId}
          onChange={e => setSelectedTreeId(e.target.value)}
          disabled={!selectedTenantId}
          style={{ ...selectStyle, opacity: selectedTenantId ? 1 : 0.5 }}
        >
          <option value="">Tree seç...</option>
          {trees.map(t => <option key={t.orgt_id} value={t.orgt_id}>{t.name}</option>)}
        </select>
        {loadError && <span style={{ color: '#f87171', fontSize: 12 }}>{loadError}</span>}
      </div>
      <div style={{ flex: 1 }}>
        <ReactFlow
          nodes={nodes}
          edges={edges}
          onNodesChange={onNodesChange}
          onEdgesChange={onEdgesChange}
          onNodeClick={handleNodeClick}
          fitView
          proOptions={{ hideAttribution: true }}
        >
          <Background color="#1e3a5f" gap={16} />
          <Controls style={{ background: '#1e293b', border: '1px solid #334155' }} />
        </ReactFlow>
      </div>
    </div>
  );
}
```

```tsx
import { useState } from 'react';
import type { Orgu } from './types';
import { traverse } from './api';
import { parseQuery } from './queryParser';

interface Props {
  query: string;
  setQuery: (q: string) => void;
  setHighlightIds: (ids: Set<string>) => void;
}

export default function RightPanel({ query, setQuery, setHighlightIds }: Props) {
  const [results, setResults] = useState<Orgu[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  const handleRun = async () => {
    setError(null);
    setResults([]);
    setHighlightIds(new Set());

    const parsed = parseQuery(query);
    if ('error' in parsed) {
      setError(parsed.error);
      return;
    }

    setLoading(true);
    try {
      const data = await traverse(parsed.selfId, parsed.expr);
      setResults(data);
      setHighlightIds(new Set(data.map(o => o.orgu_id)));
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : String(e));
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{ height: '100%', display: 'flex', flexDirection: 'column', padding: 12, gap: 10 }}>
      <textarea
        value={query}
        onChange={e => setQuery(e.target.value)}
        placeholder={'self = <uuid>\nparent'}
        spellCheck={false}
        style={{
          flexShrink: 0,
          height: 100,
          background: '#1e293b',
          color: '#e2e8f0',
          border: '1px solid #475569',
          borderRadius: 4,
          padding: 10,
          fontFamily: '"Courier New", monospace',
          fontSize: 13,
          resize: 'vertical',
          outline: 'none',
        }}
      />
      <button
        onClick={handleRun}
        disabled={loading}
        style={{
          alignSelf: 'flex-start',
          background: loading ? '#334155' : '#2563eb',
          color: '#fff',
          border: 'none',
          borderRadius: 4,
          padding: '6px 18px',
          cursor: loading ? 'not-allowed' : 'pointer',
          fontFamily: '"Courier New", monospace',
          fontSize: 13,
          letterSpacing: 1,
        }}
      >
        {loading ? '...' : 'Run ▶'}
      </button>

      {error && (
        <div style={{ color: '#f87171', fontSize: 12, background: '#1e293b', padding: '6px 8px', borderRadius: 4, border: '1px solid #f87171' }}>
          {error}
        </div>
      )}

      <div style={{ flex: 1, overflow: 'auto' }}>
        {results.length === 0 && !error && !loading && (
          <div style={{ color: '#475569', fontSize: 12, paddingTop: 4 }}>
            Sorgu çalıştır → sonuçlar burada görünür
          </div>
        )}
        {results.map(o => (
          <div
            key={o.orgu_id}
            style={{
              padding: '7px 10px',
              marginBottom: 6,
              background: '#1e293b',
              border: '1px solid #fbbf24',
              borderRadius: 4,
            }}
          >
            <div style={{ color: '#fbbf24', fontWeight: 'bold', fontSize: 13 }}>{o.name}</div>
            <div style={{ color: '#64748b', fontSize: 11, marginTop: 2 }}>
              [{o.orgu_t}] {o.orgu_id}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```
